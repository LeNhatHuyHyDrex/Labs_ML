7def softmax(z):
8 exp_z = np.exp(z - np.max(z, axis=1, keepdims=True))
9 return exp_z / np.sum(exp_z, axis=1, keepdims=True)
10
11def one_hot_encode(y, n_classes):
12 one_hot = np.zeros((len(y), n_classes))
13 one_hot[np.arange(len(y)), y] = 1
14 return one_hot
15
16class LogisticRegressionScratch:
17 """Softmax Logistic Regression viet tay cho phan loai nhieu lyp."""
18 def __init__(self, learning_rate=0.15, n_iterations=2500, l2=0.001):
19 self.lr = learning_rate
20 self.n_iter = n_iterations
21 self.l2 = l2
22 self.losses = []
23
24 def fit(self, X, y):
25 m, n = X.shape
26 n_classes = len(np.unique(y))
27 self.weights = np.zeros((n, n_classes))
28 self.bias = np.zeros((1, n_classes))
29 y_encoded = one_hot_encode(y, n_classes)
30 for i in range(self.n_iter):
31 scores = X.dot(self.weights) + self.bias
32 y_pred = softmax(scores)
33 error = y_pred - y_encoded
34 dw = X.T.dot(error) / m + self.l2 * self.weights
35 db = np.sum(error, axis=0, keepdims=True) / m
36 self.weights -= self.lr * dw
37 self.bias -= self.lr * db
38 if i % 50 == 0:
39 loss = -np.mean(np.sum(y_encoded * np.log(y_pred + 1e-12), axis=1))
40 self.losses.append(float(loss))
41 return self
42
43 def predict(self, X):
44 return np.argmax(softmax(X.dot(self.weights) + self.bias), axis=1)
45

7class NaiveBayesScratch:
8 """Multinomial Naive Bayes viet tay cho text classification."""
9
10 def __init__(self, alpha=1.0):
11 """alpha: Laplace smoothing parameter."""
12 self.alpha = alpha
13 self.class_priors = {} # P(class)
14 self.word_probs = {} # P(word | class)
15 self.vocabulary = [] # Danh sach tu vung
16 self.classes = []
17
18 def _tokenize(self, text):
19 """Tach van ban thanh cac tu (don gian: lowercase + split)."""
20 return text.lower().split()
21
22 def _build_vocabulary(self, texts):
23 """Xay dung tu dien tu tat ca van ban."""
24 vocab = set()
25 for text in texts:
26 vocab.update(self._tokenize(text))
27 self.vocabulary = sorted(vocab)
28
29 def _text_to_vector(self, text):
30 """Chuyen van ban thanh vector dem tu (Bag of Words) THU CONG."""
31 tokens = self._tokenize(text)
32 vector = np.zeros(len(self.vocabulary))
33 for token in tokens:
34 if token in self.vocabulary:
35 idx = self.vocabulary.index(token)
36 vector[idx] += 1
37 return vector
38
39 def fit(self, texts, labels):
40 """Train mo hinh Naive Bayes."""
41 self._build_vocabulary(texts)
42 self.classes = list(set(labels))
43 n_docs = len(texts)
44 n_vocab = len(self.vocabulary)
45
46 for c in self.classes:
47 # Tinh P(class) - Prior probability
48 class_docs = [texts[i] for i in range(n_docs) if labels[i] == c]
49 self.class_priors[c] = len(class_docs) / n_docs
50
51 # Dem so lan xuat hien cua moi tu trong class
52 word_counts = np.zeros(n_vocab)
53 for doc in class_docs:
54 word_counts += self._text_to_vector(doc)
55
56 # Tinh P(word | class) voi Laplace smoothing
57 total_words = np.sum(word_counts)
58 self.word_probs[c] = (word_counts + self.alpha) / (total_words + self.alpha * n_vocab)
59
60 def predict(self, texts):
61 """Du doan nhan cho danh sach van ban."""
62 predictions = []
63 for text in texts:
64 vector = self._text_to_vector(text)
65
66 # Tinh log P(class | text) cho moi class
67 class_scores = {}
68 for c in self.classes:
69 # log P(class) + sum(log P(word_i | class) * count_i)
70 log_prior = np.log(self.class_priors[c])
71 log_likelihood = np.sum(vector * np.log(self.word_probs[c]))
72 class_scores[c] = log_prior + log_likelihood
73
74 # Chon class co score cao nhat
75 predictions.append(max(class_scores, key=class_scores.get))
76
77 return predictions

6class TreeNode:
7 """Node trong Decision Tree."""
8 def __init__(self, feature_idx=None, threshold=None, left=None, right=None, value=None, depth=0):
9 self.feature_idx = feature_idx # Index cua feature de split
10 self.threshold = threshold # Nguong split
11 self.left = left # Node con trai
12 self.right = right # Node con phai
13 self.value = value # Gia tri du doan (cho leaf node)

16class DecisionTreeScratch:
17 """Decision Tree Classifier viet tay.
18 Su dung Gini Impurity de chon best split.
19 """
20
21 def __init__(self, max_depth=3, min_samples_split=2):
22 self.max_depth = max_depth
23 self.min_samples_split = min_samples_split
24 self.root = None
25 self.n_classes = None
26 self.feature_names = None
27 self.class_names = None
28
29 def _gini(self, y):
30 """Tinh Gini Impurity.
31 Gini = 1 - sum(p_i^2)
32 """
33 if len(y) == 0:
34 return 0
35 _, counts = np.unique(y, return_counts=True)
36 probs = counts / len(y)
37 return 1 - np.sum(probs ** 2)
38
39 def _best_split(self, X, y):
40 """Tim best feature va threshold de split."""
41 m, n = X.shape
42 best_gini = float('inf')
43 best_feature = None
44 best_threshold = None
45
46 for feature in range(n):
47 thresholds = np.unique(X[:, feature])
48 for threshold in thresholds:
49 left_mask = X[:, feature] <= threshold
50 right_mask = ~left_mask
51
52 if np.sum(left_mask) == 0 or np.sum(right_mask) == 0:
53 continue
54
55 # Weighted Gini
56 gini = (np.sum(left_mask) * self._gini(y[left_mask]) +
57 np.sum(right_mask) * self._gini(y[right_mask])) / m
58
59 if gini < best_gini:
60 best_gini = gini
61 best_feature = feature
62 best_threshold = threshold
63
64 return best_feature, best_threshold
65
66 def _build_tree(self, X, y, depth=0):
67 """Xay dung cay de quy."""
68 # Dieu kien dung
69 if (depth >= self.max_depth or
70 len(y) < self.min_samples_split or
71 len(np.unique(y)) == 1):
72 # Leaf node: chon class co so luong nhieu nhat
73 unique, counts = np.unique(y, return_counts=True)
74 value = unique[np.argmax(counts)]
75 # Luu ca phan bo class cho visualization
76 class_counts = np.zeros(self.n_classes)
77 for u, c in zip(unique, counts):
78 class_counts[u] = c
79 return TreeNode(value=value, depth=depth)
80
81 # Tim best split
82 best_feature, best_threshold = self._best_split(X, y)
83
84 if best_feature is None:
85 unique, counts = np.unique(y, return_counts=True)
86 return TreeNode(value=unique[np.argmax(counts)], depth=depth)
87
88 # Split data
89 left_mask = X[:, best_feature] <= best_threshold
90 right_mask = ~left_mask
91
92 # De quy xay trai va phai
93 left = self._build_tree(X[left_mask], y[left_mask], depth + 1)
94 right = self._build_tree(X[right_mask], y[right_mask], depth + 1)
95
96 return TreeNode(feature_idx=best_feature, threshold=best_threshold,
97 left=left, right=right, depth=depth)
98
99 def fit(self, X, y):
100 self.n_classes = len(np.unique(y))
101 self.root = self._build_tree(np.array(X), np.array(y))
102

111 def predict(self, X):

114 def score(self, X, y):

Code gốc của cell này - lấy đúng từ notebook (108 dòng)
1# ============================================================
2# SVM VIET TAY (FROM SCRATCH) voi RBF Kernel
3# Khong dung sklearn.svm.SVC
4# Su dung Gradient Descent tren Hinge Loss + Kernel Trick
5# ============================================================

7class SVMScratch:
8 """Support Vector Machine viet tay voi RBF Kernel.
9 Su dung Gradient Descent tren soft-margin SVM (Hinge Loss).
10
11 Primal form with kernel approximation:
12 - Tinh kernel matrix K
13 - Toi uu: min (1/2)||w||^2 + C * sum(max(0, 1 - y_i * f(x_i)))
14 """
15
16 def __init__(self, C=1.0, gamma=1.0, learning_rate=0.001, n_iterations=1000):
17 self.C = C # Regularization parameter
18 self.gamma = gamma # RBF kernel parameter
19 self.lr = learning_rate
20 self.n_iter = n_iterations
21 self.alpha = None # Dual coefficients
22 self.b = 0 # Bias
23 self.X_train = None
24 self.y_train = None
25
26 def _rbf_kernel(self, X1, X2):
27 """Tinh RBF Kernel Matrix.
28 K(x_i, x_j) = exp(-gamma * ||x_i - x_j||^2)
29 """
30 # Tinh ||x_i - x_j||^2 = ||x_i||^2 + ||x_j||^2 - 2*x_i.x_j
31 X1_sq = np.sum(X1 ** 2, axis=1).reshape(-1, 1)
32 X2_sq = np.sum(X2 ** 2, axis=1).reshape(1, -1)
33 sq_dists = X1_sq + X2_sq - 2 * X1.dot(X2.T)
34 return np.exp(-self.gamma * sq_dists)
35
36 def fit(self, X, y):
37 """Train SVM bang Gradient Descent tren dual-like form."""
38 self.X_train = np.array(X)
39 # Chuyen labels thanh -1 va +1
40 self.y_train = np.where(np.array(y) == 0, -1, 1).astype(float)
41 m = len(self.y_train)
42
43 # Tinh Kernel Matrix
44 K = self._rbf_kernel(self.X_train, self.X_train)
45
46 # Khoi tao alpha (dual coefficients)
47 self.alpha = np.zeros(m)
48 self.b = 0
49
50 # Gradient Descent
51 for _ in range(self.n_iter):
52 # Tinh decision function: f(x) = sum(alpha_i * y_i * K(x_i, x)) + b
53 decision = (self.alpha * self.y_train).dot(K) + self.b
54
55 # Tinh margins
56 margins = self.y_train * decision
57
58 # Gradient cho moi sample
59 for i in range(m):
60 if margins[i] < 1: # Misclassified hoac trong margin
61 self.alpha[i] += self.lr * (self.C * self.y_train[i] * K[i].dot(self.alpha * self.y_train) - self.alpha[i])
62 self.b += self.lr * self.C * self.y_train[i]
63 else:
64 self.alpha[i] -= self.lr * self.alpha[i] * 0.01 # Nhe regularization
65
66 # Clip alpha
67 self.alpha = np.clip(self.alpha, 0, self.C)
68
69 def predict(self, X):
70 """Du doan class."""
71 K = self._rbf_kernel(np.array(X), self.X_train)
72 decision = K.dot(self.alpha * self.y_train) + self.b
73 return np.where(decision >= 0, 1, 0)

11class KMeansScratch:
12 """K-Means Clustering viet tay hoan toan."""
13 def __init__(self, n_clusters=4, max_iter=300, random_state=42):
14 self.n_clusters = n_clusters
15 self.max_iter = max_iter
16 self.random_state = random_state
17 self.centroids = None
18 self.labels_ = None
19 def _init_centroids(self, X):
20 rng = np.random.default_rng(self.random_state)
21 indices = rng.choice(len(X), self.n_clusters, replace=False)
22 return X[indices].copy()
23 def _assign_clusters(self, X):
24 distances = np.sqrt(((X[:, None, :] - self.centroids[None, :, :]) ** 2).sum(axis=2))
25 return np.argmin(distances, axis=1)
26 def _update_centroids(self, X, labels):
27 new_centroids = np.zeros((self.n_clusters, X.shape[1]))
28 for k in range(self.n_clusters):
29 cluster_points = X[labels == k]
30 new_centroids[k] = cluster_points.mean(axis=0) if len(cluster_points) else self.centroids[k]
31 return new_centroids
32 def fit_predict(self, X):
33 self.centroids = self._init_centroids(X)
34 for iteration in range(self.max_iter):
35 labels = self._assign_clusters(X)
36 new_centroids = self._update_centroids(X, labels)
37 if np.allclose(self.centroids, new_centroids):
38 break
39 self.centroids = new_centroids
40 self.labels_ = labels
41 self.inertia_ = float(np.sum((X - self.centroids[labels]) ** 2))
42 return labels

6def make_two_moons_np(n_samples=300, noise=0.05, random_state=42):
7 """Tao du lieu hinh mat trang bang NumPy, khong dung make_moons."""
8 rng = np.random.default_rng(random_state)
9 n1 = n_samples // 2
10 n2 = n_samples - n1
11 t1 = rng.uniform(0, np.pi, n1)
12 t2 = rng.uniform(0, np.pi, n2)
13 moon1 = np.c_[np.cos(t1), np.sin(t1)]
14 moon2 = np.c_[1 - np.cos(t2), 0.5 - np.sin(t2)]
15 X = np.vstack([moon1, moon2]) + rng.normal(0, noise, size=(n_samples, 2))
16 return X[rng.permutation(n_samples)]
17
18class MeanShiftScratch:
19 """Mean Shift viet tay: dech tung diem ve trung banh lyn can trong bandwidth."""
20 def __init__(self, bandwidth=0.38, max_iter=80, tol=1e-3):
21 self.bandwidth = bandwidth
22 self.max_iter = max_iter
23 self.tol = tol
24 def fit_predict(self, X):
25 centers = X.copy()
26 for _ in range(self.max_iter):
27 new_centers = []
28 for center in centers:
29 distances = np.sqrt(np.sum((X - center) ** 2, axis=1))
30 neighbors = X[distances <= self.bandwidth]
31 new_centers.append(neighbors.mean(axis=0) if len(neighbors) else center)
32 new_centers = np.array(new_centers)
33 if np.max(np.linalg.norm(new_centers - centers, axis=1)) < self.tol:
34 break
35 centers = new_centers
36 unique_centers = []
37 labels = np.full(len(X), -1)
38 for i, center in enumerate(centers):
39 assigned = False
40 for k, uc in enumerate(unique_centers):
41 if np.linalg.norm(center - uc) < self.bandwidth / 2:
42 labels[i] = k
43 assigned = True
44 break
45 if not assigned:
46 unique_centers.append(center)
47 labels[i] = len(unique_centers) - 1
48 self.cluster_centers_ = np.array(unique_centers)
49 self.labels_ = labels
50 return labels

52class DBSCANScratch:
53 """DBSCAN Clustering viet tay hoan toan."""
54 def __init__(self, eps=0.18, min_samples=5):
55 self.eps = eps
56 self.min_samples = min_samples
57 self.labels_ = None
58 def _get_neighbors(self, X, idx):
59 distances = np.sqrt(np.sum((X - X[idx]) ** 2, axis=1))
60 return np.where(distances <= self.eps)[0]
61 def fit_predict(self, X):
62 n = len(X)
63 labels = np.full(n, -1)
64 visited = np.zeros(n, dtype=bool)
65 cluster_id = 0
66 for i in range(n):
67 if visited[i]:
68 continue
69 visited[i] = True
70 neighbors = list(self._get_neighbors(X, i))
71 if len(neighbors) < self.min_samples:
72 labels[i] = -1
73 continue
74 labels[i] = cluster_id
75 j = 0
76 while j < len(neighbors):
77 p = neighbors[j]
78 if not visited[p]:
79 visited[p] = True
80 p_neighbors = list(self._get_neighbors(X, p))
81 if len(p_neighbors) >= self.min_samples:
82 neighbors.extend([q for q in p_neighbors if q not in neighbors])
83 if labels[p] == -1:
84 labels[p] = cluster_id
85 j += 1
86 cluster_id += 1
87 self.labels_ = labels
88 return labels

3# Khong dung sklearn.mixture.GaussianMixture
4# Su dung EM Algorithm (Expectation-Maximization)

7class GMMScratch:
8 """Gaussian Mixture Model viet tay.
9 Su dung EM algorithm de fit.
10
11 E-step: Tinh trach nhiem (responsibility) cua moi component doi voi moi diem
12 M-step: Cap nhat weights, means, covariances
13 """
14
15 def __init__(self, n_components=1, max_iter=100, tol=1e-6, random_state=42):
16 self.n_components = n_components
17 self.max_iter = max_iter
18 self.tol = tol
19 self.random_state = random_state
20 self.weights = None # pi_k
21 self.means = None # mu_k
22 self.covariances = None # Sigma_k
23
24 def _multivariate_gaussian(self, X, mean, cov):
25 """Tinh mat do xac suat cua phan phoi Gaussian da chieu.
26 p(x) = (1/((2*pi)^(d/2) * |Sigma|^(1/2))) * exp(-0.5 * (x-mu)^T * Sigma^(-1) * (x-mu))
27 """
28 d = len(mean)
29 diff = X - mean
30 cov_inv = np.linalg.inv(cov + 1e-6 * np.eye(d)) # Regularization
31 cov_det = np.linalg.det(cov + 1e-6 * np.eye(d))
32
33 norm_const = 1.0 / (np.sqrt((2 * np.pi) ** d * np.abs(cov_det)))
34 exponent = -0.5 * np.sum(diff.dot(cov_inv) * diff, axis=1)
35
36 return norm_const * np.exp(exponent)
37
38 def fit(self, X):
39 """Fit GMM bang EM algorithm."""
40 np.random.seed(self.random_state)
41 n, d = X.shape
42
43 # Khoi tao parameters
44 self.weights = np.ones(self.n_components) / self.n_components
45 # Khoi tao means ngau nhien tu data
46 indices = np.random.choice(n, self.n_components, replace=False)
47 self.means = X[indices].copy()
48 # Khoi tao covariances = identity matrix
49 self.covariances = np.array([np.eye(d) for _ in range(self.n_components)])
50
51 prev_log_likelihood = -np.inf
52
53 for iteration in range(self.max_iter):
54 # E-step: Tinh responsibilities
55 responsibilities = np.zeros((n, self.n_components))
56 for k in range(self.n_components):
57 responsibilities[:, k] = self.weights[k] * self._multivariate_gaussian(
58 X, self.means[k], self.covariances[k])
59
60 # Normalize responsibilities
61 resp_sum = responsibilities.sum(axis=1, keepdims=True)
62 resp_sum = np.maximum(resp_sum, 1e-300) # Tranh chia cho 0
63 responsibilities /= resp_sum
64
65 # M-step: Cap nhat parameters
66 for k in range(self.n_components):
67 resp_k = responsibilities[:, k]
68 N_k = resp_k.sum()
69
70 if N_k < 1e-10:
71 continue
72
73 # Cap nhat weight
74 self.weights[k] = N_k / n
75
76 # Cap nhat mean
77 self.means[k] = (resp_k[:, np.newaxis] * X).sum(axis=0) / N_k
78
79 # Cap nhat covariance
80 diff = X - self.means[k]
81 self.covariances[k] = (resp_k[:, np.newaxis] * diff).T.dot(diff) / N_k
82
83 # Tinh log-likelihood de kiem tra hoi tu
84 log_likelihood = np.sum(np.log(resp_sum))
85 if abs(log_likelihood - prev_log_likelihood) < self.tol:
86 break
87 prev_log_likelihood = log_likelihood
88
89 def score_samples(self, X):
90 """Tinh log-likelihood cho moi sample."""
91 n = X.shape[0]
92 likelihoods = np.zeros(n)
93
94 for k in range(self.n_components):
95 likelihoods += self.weights[k] * self._multivariate_gaussian(
96 X, self.means[k], self.covariances[k])
97
98 return np.log(np.maximum(likelihoods, 1e-300))
