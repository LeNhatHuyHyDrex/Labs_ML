def softmax(z):
    exp_z = np.exp(z - np.max(z, axis=1, keepdims=True))
    return exp_z / np.sum(exp_z, axis=1, keepdims=True)

def one_hot_encode(y, n_classes):
    one_hot = np.zeros((len(y), n_classes))
    one_hot[np.arange(len(y)), y] = 1
    return one_hot

class LogisticRegressionScratch:
    """Softmax Logistic Regression viet tay cho phan loai nhieu lyp."""
    def __init__(self, learning_rate=0.15, n_iterations=2500, l2=0.001):
        self.lr = learning_rate
        self.n_iter = n_iterations
        self.l2 = l2
        self.losses = []

    def fit(self, X, y):
        m, n = X.shape
        n_classes = len(np.unique(y))
        self.weights = np.zeros((n, n_classes))
        self.bias = np.zeros((1, n_classes))
        y_encoded = one_hot_encode(y, n_classes)
        for i in range(self.n_iter):
            scores = X.dot(self.weights) + self.bias
            y_pred = softmax(scores)
            error = y_pred - y_encoded
            dw = X.T.dot(error) / m + self.l2 * self.weights
            db = np.sum(error, axis=0, keepdims=True) / m
            self.weights -= self.lr * dw
            self.bias -= self.lr * db
            if i % 50 == 0:
                loss = -np.mean(np.sum(y_encoded * np.log(y_pred + 1e-12), axis=1))
                self.losses.append(float(loss))
        return self

    def predict(self, X):
        return np.argmax(softmax(X.dot(self.weights) + self.bias), axis=1)

def accuracy_score_manual(y_true, y_pred):
    return np.mean(y_true == y_pred)

def classification_report_manual(y_true, y_pred, target_names):
    print(f"{'class':>18} {'precision':>10} {'recall':>10} {'f1-score':>10} {'support':>10}")
    for c, name in enumerate(target_names):
        tp = np.sum((y_pred == c) & (y_true == c))
        fp = np.sum((y_pred == c) & (y_true != c))
        fn = np.sum((y_pred != c) & (y_true == c))
        precision = tp / (tp + fp) if (tp + fp) else 0
        recall = tp / (tp + fn) if (tp + fn) else 0
        f1 = 2 * precision * recall / (precision + recall) if (precision + recall) else 0
        support = np.sum(y_true == c)
        print(f"{name:>18} {precision:>10.2f} {recall:>10.2f} {f1:>10.2f} {support:>10}")
    print("accuracy:", accuracy_score_manual(y_true, y_pred))

def train_test_split_np(X, y, test_size=0.25, random_state=42):
    rng = np.random.default_rng(random_state)
    train_parts, test_parts = [], []
    for cls in np.unique(y):
        idx = np.where(y == cls)[0]
        rng.shuffle(idx)
        n_test = max(1, int(len(idx) * test_size))
        test_parts.append(idx[:n_test])
        train_parts.append(idx[n_test:])
    train_idx = np.concatenate(train_parts)
    test_idx = np.concatenate(test_parts)
    rng.shuffle(train_idx)
    rng.shuffle(test_idx)
    return X[train_idx], X[test_idx], y[train_idx], y[test_idx]

def standardize_train_test(X_train, X_test):

    std[std == 0] = 1
    return (X_train - mean) / std, (X_test - mean) / std

# Logistic Regression viet tay, co tuy chn class_weight='balanced'.
class LogisticRegressionScratch:
    def __init__(self, learning_rate=0.08, n_epochs=500, class_weight=None):
        self.learning_rate = learning_rate
        self.n_epochs = n_epochs
        self.class_weight = class_weight
        self.losses = []

    def fit(self, X, y):
        X_b = np.c_[np.ones((len(X), 1)), X]
        self.theta = np.zeros(X_b.shape[1])
        if self.class_weight == "balanced":
            counts = np.bincount(y, minlength=2)
            class_weights = len(y) / (2 * counts)
            sample_weight = class_weights[y]
        else:
            sample_weight = np.ones(len(y))
        for epoch in range(self.n_epochs):
            proba = sigmoid(X_b @ self.theta)
            error = (proba - y) * sample_weight
            grad = X_b.T @ error / len(y)
            self.theta -= self.learning_rate * grad
            if epoch % 20 == 0:
                loss = -np.mean(sample_weight * (y*np.log(proba+1e-12) + (1-y)*np.log(1-proba+1e-12)))
                self.losses.append(float(loss))
        return self

    def predict_proba(self, X):
        return sigmoid(np.c_[np.ones((len(X), 1)), X] @ self.theta)

    def predict(self, X, threshold=0.5):
        return (self.predict_proba(X) >= threshold).astype(int)

# Metric viet tay: confusion matrix, precision, recall, f1.
def classification_metrics(y_true, y_pred):
    tp = int(np.sum((y_true == 1) & (y_pred == 1)))
    tn = int(np.sum((y_true == 0) & (y_pred == 0)))
    fp = int(np.sum((y_true == 0) & (y_pred == 1)))
    fn = int(np.sum((y_true == 1) & (y_pred == 0)))
    acc = (tp + tn) / len(y_true)
    precision = tp / max(tp + fp, 1)
    recall = tp / max(tp + fn, 1)
    f1 = 2 * precision * recall / max(precision + recall, 1e-12)
    return {"tp": tp, "tn": tn, "fp": fp, "fn": fn, "accuracy": acc, "precision": precision, "recall": recall, "f1": f1}

# Tu tinh ROC ve Precision-Recall theo nhieu threshold.
def roc_pr_points(y_true, scores):
    thresholds = np.r_[np.inf, np.sort(scores)[::-1], -np.inf]
    fpr, tpr, precision, recall = [], [], [], []
    for th in thresholds:
        pred = (scores >= th).astype(int)
        m = classification_metrics(y_true, pred)
        fpr.append(m["fp"] / max(m["fp"] + m["tn"], 1))
        tpr.append(m["recall"])
        precision.append(m["precision"])
        recall.append(m["recall"])
    return np.array(fpr), np.array(tpr), np.array(recall), np.array(precision)
